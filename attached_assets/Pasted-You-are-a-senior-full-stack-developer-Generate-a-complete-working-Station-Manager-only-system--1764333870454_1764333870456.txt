--- FIRESTORE RULES (manager-only, copy as firestore.rules)
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Owners collection: only owner can manage their profile
    match /owners/{ownerId} {
      allow create: if request.auth != null && request.auth.uid == ownerId;
      allow read: if request.auth != null && request.auth.uid == ownerId;
      allow update, delete: if request.auth != null && request.auth.uid == ownerId;
    }

    // Stations: ONLY owner or admin can read/write
    match /stations/{stationId} {
      allow create: if request.auth != null && request.resource.data.ownerId == request.auth.uid;
      allow read, update, delete: if request.auth != null &&
        (request.auth.uid == resource.data.ownerId || exists(/databases/$(database)/documents/owners/$(request.auth.uid)) && get(/databases/$(database)/documents/owners/$(request.auth.uid)).data.role == 'admin');
    }

    // Optional bookings collection restricted to station owners and booking users as needed
    match /bookings/{bookingId} {
      allow read: if false; // disable or restrict as per need
      allow write: if false;
    }
  }
}

--- STORAGE RULES (storage.rules)
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Station images: only owner or admin can write/read
    match /stations/{stationId}/images/{allPaths=**} {
      allow read: if request.auth != null && (
        request.auth.uid == get(/databases/$(database)/documents/stations/$(stationId)).data.ownerId
        || exists(/databases/$(database)/documents/owners/$(request.auth.uid)) && get(/databases/$(database)/documents/owners/$(request.auth.uid)).data.role == 'admin'
      );
      allow write: if request.auth != null && request.auth.uid == get(/databases/$(database)/documents/stations/$(stationId)).data.ownerId;
    }
  }
}
